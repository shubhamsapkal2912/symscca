def classify_numbers(limit):
    """Loop 1..limit, classify each number as even/odd, return both sums."""
    even_sum = 0
    odd_sum = 0
    for n in range(1, limit + 1):
        if n % 2 == 0:
            print(f"{n} is even")
            even_sum += n
        else:
            print(f"{n} is odd")
            odd_sum += n
    return even_sum, odd_sum


if __name__ == "__main__":
    e, o = classify_numbers(10)
    print("Sum of even numbers:", e)
    print("Sum of odd numbers :", o)

import re


def validate_contact(name, email, message, phone=""):
    """Returns the message the contact form displays."""
    errors = []
    if not name.strip():
        errors.append("Name is required")
    elif not re.fullmatch(r"[A-Za-z ]{2,50}", name.strip()):
        errors.append("Invalid name")
    if not email.strip():
        errors.append("Email is required")
    elif not re.fullmatch(r"[\w.+-]+@[\w-]+(\.[\w-]+)*\.[A-Za-z]{2,}", email.strip()):
        errors.append("Invalid email")
    if not message.strip():
        errors.append("Message is required")
    elif len(message) > 500:
        errors.append("Message too long")
    if phone and not re.fullmatch(r"[6-9]\d{9}", phone):
        errors.append("Invalid phone")
    return "Form submitted successfully" if not errors else "; ".join(errors)

def password_strength(pw):
    if len(pw) < 8:                           # decision 1

    has_upper = has_digit = has_special = False

import re


def validate_contact(name, email, message, phone=""):
    """Returns the message the contact form displays."""
    errors = []
    if not name.strip():
        errors.append("Name is required")
    elif not re.fullmatch(r"[A-Za-z ]{2,50}", name.strip()):
        errors.append("Invalid name")
    if not email.strip():
        errors.append("Email is required")
    elif not re.fullmatch(r"[\w.+-]+@[\w-]+(\.[\w-]+)*\.[A-Za-z]{2,}", email.strip()):
        errors.append("Invalid email")
    if not message.strip():
        errors.append("Message is required")
    elif len(message) > 500:
        errors.append("Message too long")
    if phone and not re.fullmatch(r"[6-9]\d{9}", phone):
        errors.append("Invalid phone")
    return "Form submitted successfully" if not errors else "; ".join(errors)
