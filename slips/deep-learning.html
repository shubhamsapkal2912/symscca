Q2

Implement a CNN model to classify CIFAR-10 dataset images.

Answer

s1q2.py
import tensorflow as tf
from tensorflow.keras import layers, models

# Load CIFAR-10 dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.cifar10.load_data()

# Normalize images
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

model = models.Sequential([
    layers.Conv2D(32, (3, 3), activation="relu", input_shape=(32, 32, 3)),
    layers.MaxPooling2D((2, 2)),

    layers.Conv2D(64, (3, 3), activation="relu"),
    layers.MaxPooling2D((2, 2)),

    layers.Flatten(),

    layers.Dense(128, activation="relu"),
    layers.Dense(10, activation="softmax")
])

model.compile(
    optimizer="adam",
    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

model.summary()

model.fit(x_train, y_train, epochs=5, batch_size=128)

loss, accuracy = model.evaluate(x_test, y_test)

print("Test Accuracy:", accuracy * 100, "%")

import numpy as np

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

def tanh(x):
    return (np.exp(x) - np.exp(-x)) / (np.exp(x) + np.exp(-x))

# Compare sigmoid and tanh for the same inputs
def compare(x):
    print("x     | sigmoid | tanh")
    for i in x:
        print(i, "|", round(sigmoid(i), 4), "|", round(tanh(i), 4))

x = np.array([-2, -1, 0, 1, 2])
compare(x)

Q2

Create a Python program to tokenize custom movie reviews and predict sentiment.

Answer

s8q2.py
import numpy as np
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, LSTM, Dense

# Custom movie reviews (1 = positive, 0 = negative)
reviews = [
    "this movie is great",
    "i loved this film",
    "amazing acting and story",
    "this movie is terrible",
    "i hated this film",
    "boring and bad story"
]
labels = np.array([1, 1, 1, 0, 0, 0])

# Tokenize and pad
tokenizer = Tokenizer(num_words=1000)
tokenizer.fit_on_texts(reviews)
sequences = tokenizer.texts_to_sequences(reviews)
x = pad_sequences(sequences, maxlen=6)

# Build model
model = Sequential([
    Embedding(1000, 8, input_length=6),
    LSTM(8),
    Dense(1, activation='sigmoid')
])

model.compile(optimizer='adam', loss='binary_crossentropy', metrics=['accuracy'])
model.fit(x, labels, epochs=50, verbose=0)

# Predict sentiment of new custom reviews
new_reviews = ["what a great film", "bad and boring movie"]
new_seq = pad_sequences(tokenizer.texts_to_sequences(new_reviews), maxlen=6)
pred = model.predict(new_seq)

for r, p in zip(new_reviews, pred):
    print(r, "->", "Positive" if p[0] > 0.5 else "Negative", p[0])

Write a Python program to implement Depth-Limited Search (DLS) algorithm.

Answer

s14q1.py
# Depth-Limited Search (DLS)
graph = {
    'A': ['B', 'C'],
    'B': ['D', 'E'],
    'C': ['F'],
    'D': [],
    'E': [],
    'F': []
}

def dls(node, goal, limit):
    print("Visiting:", node)
    if node == goal:
        return True
    if limit <= 0:
        return False
    for child in graph[node]:
        if dls(child, goal, limit - 1):
            return True
    return False

start = 'A'
goal = 'F'
limit = 2

if dls(start, goal, limit):
    print("Goal found within depth limit")
else:
    print("Goal not found within depth limit")

def gradient_descent(x, learning_rate, epochs):
    for i in range(epochs):
        gradient = 2 * x
        x = x - learning_rate * gradient

Q2

Create a Python function that trains a multilayer perceptron on a small dataset using forward and backward passes. Output loss after each epoch.

Answer

s15q2.py
import numpy as np

def sigmoid(x):
    return 1 / (1 + np.exp(-x))

# Train MLP (2-2-1) using forward and backward passes
def train_mlp(x, y, epochs, lr):
    np.random.seed(0)
    w1 = np.random.randn(2, 2)
    b1 = np.zeros((1, 2))
    w2 = np.random.randn(2, 1)
    b2 = np.zeros((1, 1))

    for epoch in range(epochs):
        # Forward pass
        h = sigmoid(np.dot(x, w1) + b1)
        out = sigmoid(np.dot(h, w2) + b2)

        loss = np.mean((y - out) ** 2)

        # Backward pass
        d_out = (out - y) * out * (1 - out)
        d_h = np.dot(d_out, w2.T) * h * (1 - h)

        w2 -= lr * np.dot(h.T, d_out)
        b2 -= lr * np.sum(d_out, axis=0, keepdims=True)
        w1 -= lr * np.dot(x.T, d_h)
        b1 -= lr * np.sum(d_h, axis=0, keepdims=True)

        print("Epoch", epoch + 1, "Loss:", loss)

x = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
y = np.array([[0], [1], [1], [0]])

train_mlp(x, y, epochs=20, lr=0.5)

import numpy as np

# Perceptron learning rule for AND gate with step activation
def step(x):
    return 1 if x >= 0 else 0

x = np.array([[0, 0], [0, 1], [1, 0], [1, 1]])
t = np.array([0, 0, 0, 1])

w = np.zeros(2)
b = 0
lr = 0.1

for epoch in range(10):
    for i in range(len(x)):
        z = np.dot(x[i], w) + b
        y = step(z)
        error = t[i] - y
        w = w + lr * error * x[i]
        b = b + lr * error

print("x1 | x2 | t")
print("-------------")
for x1, x2 in x:
    z = (x1 * w[0]) + (x2 * w[1]) + b
    print(x1, ' |', x2, ' |', step(z))

from tensorflow.keras.layers import Embedding, SimpleRNN, LSTM, GRU, Bidirectional, Dropout, Dense
